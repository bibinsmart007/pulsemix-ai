225: async function detectBPM(url) {
226:   try {
227:     const res = await fetch(url, { headers: { "Range": "bytes=0-3000000" }});
228:     const arrayBuffer = await res.arrayBuffer();
229:     const offlineCtx = new (window.OfflineAudioContext || window.webkitOfflineAudioContext)(1, 44100 * 20, 44100);
230:     const audioBuffer = await offlineCtx.decodeAudioData(arrayBuffer);
231:     const data = audioBuffer.getChannelData(0);
232:     let max = 0;
233:     for (let i = 0; i < data.length; i++) if (Math.abs(data[i]) > max) max = Math.abs(data[i]);
234:     const threshold = max * 0.8;
235:     const peaks = [];
236:     for (let i = 0; i < data.length; i++) {
237:         if (Math.abs(data[i]) > threshold) {
238:             peaks.push(i / audioBuffer.sampleRate);
239:             i += 10000;
240:         }
241:     }
242:     const intervals = {};
243:     for (let i = 0; i < peaks.length; i++) {
244:         for (let j = 1; j < 5 && i + j < peaks.length; j++) {
245:             const diff = peaks[i + j] - peaks[i];
246:             const tempo = Math.round(60 / diff);
247:             if (tempo > 80 && tempo < 180) {
248:                 intervals[tempo] = (intervals[tempo] || 0) + 1;
249:             }
250:         }
251:     }
252:     let guessedBPM = 0, maxCount = 0;
253:     for (const t in intervals) {
254:         if (intervals[t] > maxCount) { maxCount = intervals[t]; guessedBPM = parseInt(t); }
255:     }
256:     return guessedBPM;
257:   } catch (e) {
258:     return 0;
259:   }
260: }

322:   const inBpm = TRACKS[inIdx]?.bpm || 128;

342:   bAudio.currentTime = Math.max(0, targetSkip + (aPhase / ratio));

344:   // Extend mix to 16 beats (~7.5s at 128bpm)
